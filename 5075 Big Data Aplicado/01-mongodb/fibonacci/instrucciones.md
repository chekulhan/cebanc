## Aplicacion de una consulta 

```json
db.stocks.aggregate([
  // Find the day we want to analyse
  {
    $match: {
      symbol: "SPACEX",
      date: "2026-01-09"
    }
  },

  // Calculate Fibonacci levels
  {
    $set: {
      fib382: 120 - (22 * 0.382),
      fib50: 120 - (22 * 0.5),
      fib618: 120 - (22 * 0.618)
    }
  },

  // Check whether the daily low reached a Fibonacci level
  {
    $set: {
      reached382: { $lte: ["$low", "$fib382"] },
      reached50: { $lte: ["$low", "$fib50"] },
      reached618: { $lte: ["$low", "$fib618"] },

      recovery: { $subtract: ["$close", "$low"] }
    }
  },

  {
    $project: {
      _id: 0,
      date: 1,
      low: 1,
      close: 1,
      fib382: 1,
      fib50: 1,
      fib618: 1,
      reached382: 1,
      reached50: 1,
      reached618: 1,
      recovery: 1
    }
  }
])
```
